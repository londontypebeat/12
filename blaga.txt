#include <iostream>
#include <cmath>

struct Triangle {
    double a; 
    double b; 

    double getHypotenuse() const {
        return std::sqrt(a * a + b * b);
    }
};

int main() {
    double a, b;
    
    std::cout << "Введите катет a: ";
    std::cin >> a;
    std::cout << "Введите катет b: ";
    std::cin >> b;
    
    Triangle t = {a, b}; 
    
    std::cout << "Гипотенуза равна: " << t.getHypotenuse() << std::endl;
    
    return 0;
}

задание 2.2

#include <iostream>

using namespace std;

class Biker {
private:
    double speed;
    double time;

public:
    Biker(double speed, double time) {
        this->speed = speed;
        this->time = time;
    }

    double getPosition() {
        double distance = speed * time;
        return distance - 109 * int(distance / 109);
    }
};

int main() {
    double v, t;

    cout << "Введите скорость и время: ";
    cin >> v >> t;

    Biker biker(v, t);

    cout << "Вася остановится на отметке: "
         << biker.getPosition() << " км" << endl;

    return 0;
}
